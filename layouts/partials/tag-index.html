{{- /* Renders the tag card grid plus one tag's posts expanded beneath it. ctx: page — the current page selected — RelPermalink of the term to expand; empty expands the most-used tag. Terms are matched on RelPermalink rather than the taxonomy key, because the key keeps the original spacing ("creative writing") while the URL segment is urlized ("creative-writing"). */ -}} {{- $terms := slice -}} {{- range $key, $taxonomy := site.Taxonomies.tags -}} {{- $term := site.GetPage (printf "/tags/%s" $key) -}} {{- with $term -}} {{- $count := len $taxonomy -}} {{- /* One sort key gives count descending, then name ascending. */ -}} {{- $terms = $terms | append (dict "url" .RelPermalink "name" (lower .LinkTitle) "count" $count "page" . "sortKey" (printf "%04d-%s" (sub 9999 $count) $key)) -}} {{- end -}} {{- end -}} {{- $terms = sort $terms "sortKey" "asc" -}} {{- $selected := .selected -}} {{- if and (not $selected) (gt (len $terms) 0) -}} {{- $selected = (index $terms 0).url -}} {{- end -}}
{{- range $t := $terms }} #{{ $t.name }} {{ $t.count }} {{- end }}
{{- range $t := $terms }} {{- if eq $t.url $selected }}

#{{ $t.name }} — {{ $t.count }} post{{ if ne $t.count 1 }}s{{ end }}

{{- range $t.page.Pages.ByDate.Reverse }} {{ .Title }} {{ .Date.Format "2006.01.02" }} {{- end }}
{{- end }} {{- end }}
